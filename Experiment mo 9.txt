#include <SoftwareSerial.h>

SoftwareSerial XBee(2, 3);   // RX, TX

void setup() {
  Serial.begin(9600);
  XBee.begin(9600);
}

void loop() {
  XBee.println("Hello from Arduino");
  delay(1000);
}


xbee_receive.py
import serial

xbee = serial.Serial('/dev/ttyUSB0', 9600)   # change to your XBee port

while True:
    if xbee.in_waiting > 0:
        incoming_data = xbee.readline().decode('utf-8').strip()
        print(f"Received from Arduino: {incoming_data}")
