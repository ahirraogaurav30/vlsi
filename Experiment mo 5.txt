import Adafruit_DHT as dht
import RPi.GPIO as GPIO
from time import sleep
from urllib.request import urlopen

GPIO.setwarnings(False)
GPIO.setmode(GPIO.BCM)
GPIO.setup(24, GPIO.IN)                    # MQ3 digital output

myAPI = "YOUR_WRITE_API_KEY"               # your own ThingSpeak key
baseURL = "https://api.thingspeak.com/update?api_key=%s" % myAPI

def DHT11_data():
    humi, temp = dht.read_retry(11, 23)
    return humi, temp

while True:
    humidity, temperature = DHT11_data()
    if humidity is None or temperature is None:
        print("Sensor read failed")
        sleep(2)
        continue

    print('Temp: {0:0.1f} C  Humidity: {1:0.1f} %'.format(temperature, humidity))

    if GPIO.input(24) == False:            # active LOW
        smoke = 1
        print('Smoke Detected')
    else:
        smoke = 0
        print('Smoke is not Detected')

    urlopen(baseURL + "&field1=%s&field2=%s&field3=%s" % (temperature, humidity, smoke))
    sleep(15)                              # ThingSpeak free tier: one update per 15 s
